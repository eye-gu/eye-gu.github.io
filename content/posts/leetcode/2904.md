class Solution {
        public String shortestBeautifulSubstring(String s, int k) {
            // 找第一个1
            int left = -1;
            for (int i =0; i<s.length(); i++) {
                if (s.charAt(i) == '1') {
                    left = i;
                    break;
                }
            }
            // 没找到，返回空字符串
            if (left == -1) {
                return "";
            }
            // 只要1个1，直接返回1
            if (k == 1) {
                return "1";
            }
            int right = left+1;
            int count = 1;
            int minLen = Integer.MAX_VALUE;
            String res = "";
            while (right < s.length()) {
                char c = s.charAt(right);
                // 首尾一定是1
                if (c != '1') {
                    right++;
                    continue;
                }
                count++;
                if (count == k) {
                    if (right - left + 1 < minLen) {
                        minLen = right - left + 1;
                        res = s.substring(left, right + 1);
                    } else if (right - left + 1 == minLen) {
                        res = res.compareTo(s.substring(left, right + 1)) < 0 ? res : s.substring(left, right + 1);
                    }
                    // 找到left右边的下一个1，count需减1
                    left++;
                    while (s.charAt(left) != '1') {
                        left++;
                    }
                    count--;
                }
                right++;
            }
            return res;
        }
    }
