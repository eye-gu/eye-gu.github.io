    class Solution {
        public int minOperations(String s) {
            byte[] bytes = s.getBytes();
            byte min = bytes[0];
            int minCount = 0;
            byte max = bytes[0];
            int maxCount = 0;
            boolean flag = true;
            for (int i =0; i<bytes.length; i++) {
                if (i != 0 && bytes[i] < bytes[i-1]) {
                    flag = false;
                }
                if (bytes[i] < min) {
                    min = bytes[i];
                    minCount = 1;
                } else if (bytes[i] == min) {
                    minCount++;
                }
                if (bytes[i] > max) {
                    max = bytes[i];
                    maxCount = 1;
                } else if (bytes[i] == max) {
                    maxCount++;
                }
            }
            // 已经有序
            if (flag) {
                return 0;
            }
            // 第一个是最小的字符, 只要1次: [1,n)
            if (min == bytes[0]) {
                return 1;
            }
            // 最后一个是最大的字符,只要1次: [0,n-1)
            if (max == bytes[bytes.length - 1]) {
                return 1;
            }
            // 最后一个是最小的字符
            if (min == bytes[bytes.length - 1]) {
                if (bytes.length == 2) {
                    return -1;
                }
                // 最大字符在第一个
                if (max == bytes[0]) {
                    // 最大或最小大于一个
                    // 假设最小有多个, 那么[0,n-1)排序一次后, 0位置就是最小的字符, n-1位置的最小字符还需要排序一次
                    if (maxCount > 1 || minCount > 1) {
                        return 2;
                    } else {
                        // 否则需要3次:[0,n-1)排序,最小字符在n-1,最大字符在n-2. [1,n)排序,最小字符在位置1,最大字符在最后一位. [0,n-1)排序,最小字符到0
                        return 3;
                    }
                } else {
                    // 最小字符在n-1,两次即可
                    return 2;
                }
            }
            // 最小最大不在首位和尾位,那么一次[0,n-1),最小字符到了首位, 一次[1,n),最大字符到了尾位
            return 2;
        }
    }
