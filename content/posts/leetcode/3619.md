    class Solution {
        public int countIslands(int[][] grid, int k) {
            int ans = 0;
            for (int i=0; i<grid.length; i++) {
                for (int j=0; j<grid[0].length; j++) {
                    if (grid[i][j] == 0) {
                        continue;
                    }
                    // long防止溢出
                    long sum = 0;
                    Queue<int[]> queue = new ArrayDeque<>();
                    queue.add(new int[]{i, j});
                    while (!queue.isEmpty()) {
                        int[] poll = queue.poll();
                        // 虽然加到队列的时候检查了大于0, 但是因为一个坐标会被多次入队, 所以需要再检查一次
                        if (grid[poll[0]][poll[1]] == 0) {
                            continue;
                        }
                        sum += grid[poll[0]][poll[1]];
                        grid[poll[0]][poll[1]] = 0;
                        if (poll[0] > 0 && grid[poll[0]-1][poll[1]] > 0) {
                            queue.add(new int[]{poll[0]-1, poll[1]});
                        }
                        if (poll[0] < grid.length-1 && grid[poll[0]+1][poll[1]] > 0) {
                            queue.add(new int[]{poll[0]+1, poll[1]});
                        }
                        if (poll[1] > 0 && grid[poll[0]][poll[1]-1] > 0) {
                            queue.add(new int[]{poll[0], poll[1]-1});
                        }
                        if (poll[1] < grid[0].length-1 && grid[poll[0]][poll[1]+1] > 0) {
                            queue.add(new int[]{poll[0], poll[1]+1});
                        }
                    }
                    if (sum % k == 0) {
                        ans++;
                    }
                }
            }
            return ans;
        }
    }

    class Solution {
        public int countIslands(int[][] grid, int k) {
            int ans = 0;
            for (int i=0; i<grid.length; i++) {
                for (int j=0; j<grid[0].length; j++) {
                    if (grid[i][j] == 0) {
                        continue;
                    }
                    // long防止溢出
                    long sum = dfs(grid, i, j);
                    if (sum % k == 0) {
                        ans++;
                    }
                }
            }
            return ans;
        }

        private long dfs(int[][] grid, int i, int j) {
            long sum = grid[i][j];
            grid[i][j] = 0;
            if (i > 0 && grid[i-1][j] > 0) {
                sum += dfs(grid, i-1, j);
            }
            if (i < grid.length-1 && grid[i+1][j] > 0) {
                sum += dfs(grid, i+1, j);
            }
            if (j > 0 && grid[i][j-1] > 0) {
                sum += dfs(grid, i, j-1);
            }
            if (j < grid[0].length-1 && grid[i][j+1] > 0) {
                sum += dfs(grid, i, j+1);
            }
            return sum;
        }
    }
