    class Solution {
        public long[] resultArray(int[] nums, int k) {
            long[] result = new long[k];
            // cur[x] 代表以当前元素结尾, 余数x的数量
            long[] cur = new long[k];
            for (int num : nums) {
                long[] next = new long[k];
                // 只选num
                next[num % k] = 1;
                // num和之前每种余数都相乘,构成num结尾的,余数的数量
                for (int r = 0; r < k; r++) {
                    next[(int) ((long) r * num % k)] += cur[r];
                }
                for (int r = 0; r < k; r++) {
                    result[r] += next[r];
                }
                cur = next;
            }
            return result;
        }
    }
