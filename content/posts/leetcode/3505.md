想到了dp, 来来回回提交了好几遍, 结果还是超时, 每次遍历都需要一次x长度数组的排序
```java
    class Solution {
        public long minOperations(int[] nums, int x, int k) {
            // dp[i][j] 代表 nums[0..i] 中选出 j 个 长度x的相等数组, 操作数最小值
            long[][] dp = new long[nums.length][k+1];
            for (int i = 0; i < nums.length; i++) {
                Arrays.fill(dp[i], -1);
            }
            int[] tmp = new int[x];
            int mid = x / 2;
            for (int i = x - 1; i < nums.length; i++) {

                // 把 i-x 到 i 这段数组赋值后排序,计算操作数,这里复杂度太高超时
                for (int j = 0; j < x; j++) {
                    tmp[j] = nums[i - j];
                }
                Arrays.sort(tmp);
                long operations = 0;
                for (int j=0;j<x;j++) {
                    operations += Math.abs(tmp[j] - tmp[mid]);
                }


                for (int j = 1; j <= k; j++) {
                    // x-1特殊处理
                    if (i == x-1) {
                        if (j==1) {
                            dp[i][j] = operations;
                        }
                    } else {
                        // >=0是因为dp初始化是-1.初始化成Long.MAX_VALUE/2就可以去掉
                        // dp[i - x][j - 1] >= 0 说明nums[0..i-x] 中选出 j-1 个 长度x的相等数组是可以的
                        if (dp[i - x][j - 1] >= 0) {
                            if (dp[i-1][j] >= 0) {
                                dp[i][j] = Math.min(dp[i - x][j - 1] + operations, dp[i-1][j]);
                            } else {
                                dp[i][j] = dp[i - x][j - 1] + operations;
                            }
                        } else if (j == 1) {
                            if (dp[i-1][j] >= 0) {
                                dp[i][j] = Math.min(operations, dp[i-1][j]);
                            } else {
                                dp[i][j] = operations;
                            }
                        }
                    }
                }
            }
            return dp[nums.length-1][k];
        }
    }
```

```java
class Solution {
        private int[] sorted;
        private int size;
        private int[] count;
        private long[] sum;

        public long minOperations(int[] nums, int x, int k) {
            int n = nums.length;

            sorted = nums.clone();
            Arrays.sort(sorted);
            size = 0;
            for (int v : sorted) {
                if (size == 0 || sorted[size - 1] != v) {
                    sorted[size++] = v;
                }
            }
            count = new int[size + 1];
            sum = new long[size + 1];

            long[] cost = new long[n];
            for (int i = 0; i < x; i++) {
                add(rank(nums[i]), 1, nums[i]);
            }
            cost[x - 1] = windowCost(x);
            for (int i = x; i < n; i++) {
                add(rank(nums[i - x]), -1, -nums[i - x]);
                add(rank(nums[i]), 1, nums[i]);
                cost[i] = windowCost(x);
            }

            long[][] dp = new long[n][k + 1];
            for (int i = 0; i < n; i++) {
                Arrays.fill(dp[i], -1);
            }
            for (int i = x - 1; i < n; i++) {
                for (int j = 1; j <= k; j++) {
                    if (i == x - 1) {
                        if (j == 1) {
                            dp[i][j] = cost[i];
                        }
                    } else {
                        if (dp[i - x][j - 1] >= 0) {
                            if (dp[i - 1][j] >= 0) {
                                dp[i][j] = Math.min(dp[i - x][j - 1] + cost[i], dp[i - 1][j]);
                            } else {
                                dp[i][j] = dp[i - x][j - 1] + cost[i];
                            }
                        } else if (j == 1) {
                            if (dp[i - 1][j] >= 0) {
                                dp[i][j] = Math.min(cost[i], dp[i - 1][j]);
                            } else {
                                dp[i][j] = cost[i];
                            }
                        }
                    }
                }
            }
            return dp[n - 1][k];
        }

        private int rank(int v) {
            return Arrays.binarySearch(sorted, 0, size, v) + 1;
        }

        private void add(int r, int deltaCount, long deltaSum) {
            for (int i = r; i <= size; i += i & -i) {
                count[i] += deltaCount;
                sum[i] += deltaSum;
            }
        }

        private int prefixCount(int i) {
            int s = 0;
            for (; i > 0; i -= i & -i) {
                s += count[i];
            }
            return s;
        }

        private long prefixSum(int i) {
            long s = 0;
            for (; i > 0; i -= i & -i) {
                s += sum[i];
            }
            return s;
        }

        private int kth(int k) {
            int pos = 0;
            for (int step = Integer.highestOneBit(size); step > 0; step >>= 1) {
                if (pos + step <= size && count[pos + step] < k) {
                    pos += step;
                    k -= count[pos];
                }
            }
            return pos + 1;
        }

        private long windowCost(int x) {
            int r = kth(x - x / 2);
            long median = sorted[r - 1];
            long leftCount = prefixCount(r - 1);
            long leftSum = prefixSum(r - 1);
            long rightCount = prefixCount(size) - prefixCount(r);
            long rightSum = prefixSum(size) - prefixSum(r);
            return median * leftCount - leftSum + rightSum - median * rightCount;
        }
    }
```
