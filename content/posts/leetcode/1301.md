// 动态规划
    class Solution {
        int MOD = 1000000007;
        public int[] pathsWithMaxScore(List<String> board) {
            int n = board.size();
            int[][][] dp = new int[n][n][2];
            dp[n - 1][n - 1][0] = 0; // 存路径最大和
            dp[n - 1][n - 1][1] = 1; // 存路径最大和的方案数,起点是1
            for (int i = n-1; i >= 0; i--) {
                for (int j = n-1; j >= 0; j--) {
                    if (board.get(i).charAt(j) == 'X') {
                        continue;
                    }
                    if (i == n-1 && j == n-1) {
                        continue;
                    }
                    // 找右边,下边,右下边的最大的路径和,累加其方案数
                    int max = 0;
                    int count = 0;
                    if (i < n-1) {
                        if (dp[i+1][j][0] > max) {
                            count = dp[i+1][j][1];
                        } else if (dp[i+1][j][0] == max){
                            count = (count + dp[i+1][j][1]) %MOD;
                        }
                        max = Math.max(max, dp[i+1][j][0]);
                    }
                    if (j < n-1) {
                        if (dp[i][j+1][0] > max) {
                            count = dp[i][j+1][1];
                        } else if (dp[i][j+1][0] == max) {
                            count = (count + dp[i][j+1][1]) % MOD;
                        }
                        max = Math.max(max, dp[i][j+1][0]);
                    }
                    if (i < n-1 && j < n-1) {
                        if (dp[i+1][j+1][0] > max) {
                            count = dp[i+1][j+1][1];
                        } else if (dp[i+1][j+1][0] == max) {
                            count = (count + dp[i+1][j+1][1]) % MOD;
                        }
                        max = Math.max(max, dp[i+1][j+1][0]);
                    }
                    // 如果方案数是0,那么该点不可到达
                    if (count == 0) {
                        dp[i][j][0] = 0;
                    } else {
                        if (board.get(i).charAt(j) != 'E') {
                            dp[i][j][0] = max + (board.get(i).charAt(j) - '0');
                        } else {
                            dp[i][j][0] = max;
                        }
                    }
                    dp[i][j][1] = count;
                }
            }
            return new int[]{dp[0][0][0], dp[0][0][1]};
        }
    }
