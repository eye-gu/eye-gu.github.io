class Solution {
    public int largestOverlap(int[][] img1, int[][] img2) {
        int n = img1.length;
        int[][] count = new int[2 * n][2 * n];

        // 所有的img1的1 和 所有的img2的1, 计算出它们的距离
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                if (img1[i][j] == 0) {
                    continue;
                }
                for (int k=0; k<n; k++) {
                    for (int l =0; l<n; l++) {
                        if (img2[k][l] == 0) {
                            continue;
                        }
                        count[n + i - k][n+j-l] += 1;
                    }
                }
            }
        }

        int max = 0;
        for (int i=0; i<2*n; i++) {
            for (int j=0; j<2*n; j++) {
                max = Math.max(max, count[i][j]);
            }
        }
        return max;
    }
}
