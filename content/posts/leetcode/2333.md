    class Solution {
        public long minSumSquareDiff(int[] nums1, int[] nums2, int k1, int k2) {
            int n = nums1.length;
            // 数组的diff -> diff的count
            TreeMap<Integer, Integer> diffTreeMap = new TreeMap<>();
            for (int i =0; i < n; i++) {
                int diff = Math.abs(nums1[i] - nums2[i]);
                if (diff > 0) {
                    if (diffTreeMap.containsKey(diff)) {
                        diffTreeMap.put(diff, diffTreeMap.get(diff) + 1);
                    } else {
                        diffTreeMap.put(diff, 1);
                    }
                }
            }
            if (diffTreeMap.isEmpty()) {
                return 0;
            }
            // 对nums1还是nums2操作是等价的,因此可以合并成总的操作数k
            long k = k1 + k2;
            while (k > 0 && !diffTreeMap.isEmpty()) {
                // 当前最大的diff
                Map.Entry<Integer, Integer> maxDiffEntry = diffTreeMap.pollLastEntry();
                // 第二大的diff
                int secondMaxDiff = 0;
                if (!diffTreeMap.isEmpty()) {
                    secondMaxDiff = diffTreeMap.lastKey();
                }
                int maxDiff = maxDiffEntry.getKey();
                int maxDiffCount = maxDiffEntry.getValue();
                // 计算把所有最大diff修改成第二大diff所需要的操作数
                long needK = (long) maxDiffCount * (maxDiff - secondMaxDiff);
                // 如果需要的操作数小于等于k
                if (needK <= k) {
                    k -= needK;
                    if (secondMaxDiff == 0) {
                        break;
                    }
                    // 转换成第二大diff
                    diffTreeMap.put(secondMaxDiff, diffTreeMap.get(secondMaxDiff) + maxDiffCount);
                    continue;
                }
                // 如果操作数不足, 那么平均把maxdiff往下降.
                // a是每个maxdiff下降的大小, b是maxdiff下降到maxdiff-a-1的数量
                int a = Math.toIntExact(k / maxDiffCount);
                int b = Math.toIntExact(k % maxDiffCount);

                if (maxDiff - a - 1 > 0) {
                    if (maxDiff - a - 1 == secondMaxDiff) {
                        diffTreeMap.put(secondMaxDiff, diffTreeMap.get(secondMaxDiff) + b);
                    } else {
                        diffTreeMap.put(maxDiff - a - 1, b);
                    }
                }
                diffTreeMap.put(maxDiff - a, maxDiffCount - b);
                break;
            }
            long sum = 0;
            for (Map.Entry<Integer, Integer> entry : diffTreeMap.entrySet()) {
                sum = sum + (long) entry.getKey() * entry.getKey() * entry.getValue();
            }
            return sum;
        }
    }

    class Solution {
        public long minSumSquareDiff(int[] nums1, int[] nums2, int k1, int k2) {
            long k = (long) k1 + k2;
            int n = nums1.length;

            long sum = 0;
            for (int i = 0; i < n; i++) {
                nums1[i] = Math.abs(nums1[i] - nums2[i]);
                sum += nums1[i];
            }
            if (sum <= k) {
                return 0;
            }

            Arrays.sort(nums1);
            int[] d = new int[n + 1];
            // 从大到小, d[n]=0
            for (int i = 0; i < n; i++) {
                d[i] = nums1[n - 1 - i];
            }

            for (int i = 1; i <= n; i++) {
                // [0,i-1]的值都是d[i-1], 计算把这些d[i-1]全部转换成d[i]所需的操作数
                long cost = (long) (d[i - 1] - d[i]) * i;
                if (cost > k) {
                    // q是每个d[i-1]都先转换成d[i-1] - q
                    // r是其中r个d[i-1]再减1,转换成d[i-1] - q - 1
                    long q = k / i, r = k % i, hi = d[i - 1] - q;
                    long ans = hi * hi * (i - r) + (hi - 1) * (hi - 1) * r;
                    for (int j = i; j < n; j++) {
                        ans += (long) d[j] * d[j];
                    }
                    return ans;
                }
                // 减去cost,代表[0,i-1]的d[i-1]转换成了d[i]
                k -= cost;
            }
            return 0;
        }
    }
