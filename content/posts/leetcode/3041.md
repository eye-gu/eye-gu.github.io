    class Solution {
        public int maxSelectedElements(int[] nums) {
            // 排序
            Arrays.sort(nums);
            // dp[i]代表i结尾的连续数组的最大长度。最大就是最大值+1
            int[] dp = new int[nums[nums.length - 1] + 2];
            int res = 0;
            for (int num : nums) {
                // 必须先算num+1，因为会用到dp[num]，否则就等于num自身构造了num，num+1两个连续数字
                // num变成num+1
                dp[num + 1] = Math.max(dp[num + 1], dp[num] + 1);
                // num不变
                dp[num] = Math.max(dp[num], dp[num - 1] + 1);
                res = Math.max(Math.max(res, dp[num]), dp[num + 1]);
            }
            return res;
        }
    }
