class Solution {
        public int maximumSafenessFactor(List<List<Integer>> grid) {
            int n = grid.size();
            if (grid.get(0).get(0) == 1 || grid.get(n - 1).get(n - 1) == 1) {
                return 0;
            }
            int[][] dirs = new int[][]{{1, 0}, {-1, 0}, {0, 1}, {0, -1}};

            // 使用bfs计算每个格子离小偷的最短曼哈顿距离
            Queue<int[]> queue = new ArrayDeque<>();
            for (int i = 0; i < n; i++) {
                for (int j = 0; j < n; j++) {
                    if (grid.get(i).get(j) == 1) {
                        queue.offer(new int[]{0, i, j});
                    }
                    grid.get(i).set(j, -1);
                }
            }
            while (!queue.isEmpty()) {
                int[] poll = queue.poll();
                int step = poll[0];
                int x = poll[1];
                int y = poll[2];
                if (grid.get(x).get(y) != -1) {
                    continue;
                }
                grid.get(x).set(y, step);
                for (int[] dir : dirs) {
                    int nx = x + dir[0];
                    int ny = y + dir[1];
                    if (nx < 0 || nx >= n || ny < 0 || ny >= n) {
                        continue;
                    }
                    queue.offer(new int[]{step + 1, nx, ny});
                }
            }

            // dijkstra
            // 每次都走目前可以到达的格子中,离小偷最远的那个
            boolean[][] seen = new boolean[n][n];
            seen[0][0] = true;
            // 存储所有可以到达的格子, 按照离小偷的距离排序
            PriorityQueue<int[]> pq = new PriorityQueue<>((a, b) -> b[0] - a[0]);
            pq.offer(new int[]{grid.get(0).get(0), 0, 0});
            // 记录目前路径最小的曼哈顿距离
            int min = grid.get(0).get(0);
            while (!pq.isEmpty()) {
                // 目前可以到达的格子中, 离小偷最远的格子
                int[] cur = pq.poll();
                int x = cur[1];
                int y = cur[2];
                if (x == n - 1 && y == n - 1) {
                    return min;
                }
                min = Math.min(min, cur[0]);
                for (int[] dir : dirs) {
                    int nx = x + dir[0];
                    int ny = y + dir[1];
                    if (nx < 0 || nx >= n || ny < 0 || ny >= n) {
                        continue;
                    }
                    if (seen[nx][ny]) {
                        continue;
                    }
                    seen[nx][ny] = true;
                    pq.offer(new int[]{grid.get(nx).get(ny), nx, ny});
                }
            }
            return min;
        }
    }
