class Solution {
        public int totalNumbers(int[] digits) {
            int[] count = new int[10];
            for (int digit : digits) {
                count[digit]++;
            }
            int ans = 0;
            // i 个位,必须是偶数
            for (int i=0; i<10; i+=2) {
                if (count[i] == 0) {
                    continue;
                }
                // j 百位必须大于0
                for (int j=1; j<10; j++) {
                    if (count[j] == 0) {
                        continue;
                    }
                    if (i == j && count[j] == 1)  {
                        continue;
                    }
                    // k 十位
                    for (int k=0; k< 10; k++) {
                        if (count[k] == 0) {
                            continue;
                        }
                        if (i == k && i == j && count[k] <= 2) {
                            continue;
                        }
                        if (i == k && count[k] <= 1) {
                            continue;
                        }
                        if (j == k && count[k] <= 1) {
                            continue;
                        }
                        ans += 1;
                    }
                }
            }
            return ans;
        }
    }
