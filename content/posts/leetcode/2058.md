class Solution {
        public int[] nodesBetweenCriticalPoints(ListNode head) {
            int i =0;
            // 第一个临界点的索引
            int firstIndex = -1;
            // 上一个临界点的索引
            int preIndex = -1;
            // 临界点最小距离
            int minDistance = Integer.MAX_VALUE;
            // 临界点最大距离
            int maxDistance = -1;
            // 链表上一个值
            int preVal = head.val;
            while (head.next != null) {

                if ((head.val > preVal && head.val > head.next.val)
                || head.val < preVal && head.val < head.next.val) {
                    // 第一个临界点
                    if (firstIndex == -1) {
                        firstIndex = i;
                    } else {
                        // 最大距离一定是和第一个临界点的距离
                        maxDistance = i - firstIndex;
                        // 最小距离一定是相邻的两个临界点的距离
                        minDistance = Math.min(minDistance, i - preIndex);
                    }
                    preIndex = i;
                }
                i++;
                preVal = head.val;
                head = head.next;
            }
            return new int[]{minDistance == Integer.MAX_VALUE ? -1 : minDistance, maxDistance};
        }
    }
