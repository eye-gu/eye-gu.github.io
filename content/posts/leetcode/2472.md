没有考虑到不重叠的条件, 所以错误
```java
    class Solution {
        public int maxPalindromes(String s, int k) {
            int ans = 0;
            for (int i =0; i<s.length(); i++) {
                int left = i;
                int right = i + 1;
                while (left >= 0 && right < s.length() && s.charAt(left) == s.charAt(right)) {
                    left--;
                    right++;
                }
                if (right - left - 1 >= k) {
                    ans += (right - left - 1 - k) / 2 + 1;
                }
                left = i-1;
                right=i+1;
                while (left >= 0 && right < s.length() && s.charAt(left) == s.charAt(right)) {
                    left--;
                    right++;
                }
                if (right - left - 1 >= k) {
                    ans += (right - left - 1 - k) / 2 + 1;
                }
            }
            return ans;
        }
    }
```

使用dp的方案
```java
class Solution {
        public int maxPalindromes(String s, int k) {
            int n = s.length();
            // dp[i] 代表 s[0,i-1]的不重叠回文串数量
            int[] dp = new int[n + 1];
            for (int i = 0; i < n; i++) {
                dp[i+1] = dp[i];
                if (i-k+1 < 0) {
                    continue;
                }
                // 如果 s[i-k+1,i] 是回文串, 则 dp[i+1] = dp[i-k+1] + 1
                if (isPalindrome(s, i-k+1, i)) {
                    dp[i+1] = Math.max(dp[i+1], dp[i-k+1] + 1);
                    continue;
                }
                if (i-k < 0) {
                    continue;
                }
                // 如果 s[i-k,i] 是回文串, 则 dp[i+1] = dp[i-k] + 1
                if (isPalindrome(s, i-k, i)) {
                    dp[i+1] = Math.max(dp[i+1], dp[i-k] + 1);
                }
            }
            return dp[n];
        }

        private boolean isPalindrome(String s, int left, int right) {
            if (left < 0) {
                return false;
            }
            while (left < right) {
                if (s.charAt(left) != s.charAt(right)) {
                    return false;
                }
                left++;
                right--;
            }
            return true;
        }
    }
```
