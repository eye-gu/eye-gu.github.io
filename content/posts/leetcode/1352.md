    class ProductOfNumbers {
        // 前缀乘积
        List<Integer> preProduct;
        public ProductOfNumbers() {
            preProduct = new ArrayList<>();
        }

        public void add(int num) {
            // 遇到0就全部清空
            if (num == 0) {
                preProduct.clear();
                return;
            }
            if (preProduct.isEmpty()) {
                preProduct.add(num);
            } else {
                preProduct.add(preProduct.get(preProduct.size() - 1) * num);
            }
        }

        public int getProduct(int k) {
            // 说明遇到了0
            if (k > preProduct.size()) {
                return 0;
            }
            if (k == preProduct.size()) {
                return preProduct.get(preProduct.size() - 1);
            }
            // 最后一个是所有的乘积，除以最后k个前面的乘积，即最后k个的乘积
            return preProduct.get(preProduct.size() - 1) /
            preProduct.get(preProduct.size() - k - 1);
        }
    }
