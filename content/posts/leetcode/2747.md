    class Solution {
        public int[] countServers(int n, int[][] logs, int x, int[] queries) {
            // 根据访问时间排序
            Arrays.sort(logs, new Comparator<int[]>() {
                @Override
                public int compare(int[] o1, int[] o2) {
                    return o1[1] - o2[1];
                }
            });

            // 对查询排序
            Integer[] querySortIndexes = new Integer[queries.length];
            for (int i =0; i<querySortIndexes.length; i++) {
                querySortIndexes[i] = i;
            }
            Arrays.sort(querySortIndexes, new Comparator<Integer>() {
                @Override
                public int compare(Integer o1, Integer o2) {
                    return queries[o1] - queries[o2];
                }
            });

            int[] res = new int[queries.length];
            int l = 0;
            int r = 0;
            Map<Integer, Integer> logServerCount = new HashMap<>();
            for (int i=0; i<queries.length; i++) {
                // 检查logs[r]是否在查询范围内
                while (r < logs.length && logs[r][1] <= queries[querySortIndexes[i]]) {
                    if (logServerCount.containsKey(logs[r][0])) {
                        logServerCount.put(logs[r][0], logServerCount.get(logs[r][0]) + 1);
                    } else {
                        logServerCount.put(logs[r][0], 1);
                    }
                    r++;
                }
                // 检查logs[l]是否在查询范围内
                while (l < logs.length && logs[l][1] < queries[querySortIndexes[i]] - x) {
                    Integer count = logServerCount.get(logs[l][0]);
                    if (count == 1) {
                        logServerCount.remove(logs[l][0]);
                    } else {
                        logServerCount.put(logs[l][0], count-1);
                    }
                    l++;
                }
                res[querySortIndexes[i]] =n - logServerCount.size();
            }
            return res;
        }
    }

    class Solution {
        public int[] countServers(int n, int[][] logs, int x, int[] queries) {
            // 根据访问时间排序
            Arrays.sort(logs, Comparator.comparingInt(o -> o[1]));

            // 对查询排序
            Integer[] querySortIndexes = new Integer[queries.length];
            for (int i =0; i<querySortIndexes.length; i++) {
                querySortIndexes[i] = i;
            }
            Arrays.sort(querySortIndexes, Comparator.comparingInt(o -> queries[o]));

            int[] res = new int[queries.length];
            int l = 0;
            int r = 0;
            // 服务器数量为n，用数组存查询区间内的访问记录
            int[] logServerCount = new int[n+1];
            int active = 0;
            for (int i : querySortIndexes) {
                // 检查logs[r]是否在查询范围内
                while (r < logs.length && logs[r][1] <= queries[i]) {
                    if (logServerCount[logs[r][0]] == 0) {
                        active++;
                    }
                    logServerCount[logs[r][0]]++;
                    r++;
                }
                // 检查logs[l]是否在查询范围内
                while (l < logs.length && logs[l][1] < queries[i] - x) {
                    logServerCount[logs[l][0]]--;
                    if (logServerCount[logs[l][0]] == 0) {
                        active--;
                    }
                    l++;
                }
                res[i] = n - active;
            }
            return res;
        }
    }
