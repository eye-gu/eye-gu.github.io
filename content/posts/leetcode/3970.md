```java
    class Solution {
        public int shortestPath(int n, int[][] edges, String labels, int k) {
            List[] graph = new List[n];
            for (int i = 0; i < n; i++) {
                graph[i] = new ArrayList<>();
            }
            for (int[] edge : edges) {
                graph[edge[0]].add(new int[]{edge[1], edge[2]});
            }
            // visited[i] 表示是否访问过节点 i, 避免环
            boolean[] visited = new boolean[n];
            visited[0] = true;
            int min = dfs(n, graph, labels, k, 0, 0, 1, visited);
            return min == Integer.MAX_VALUE ? -1 : min;
        }
        // cur 当前节点，sum 当前路径长度，repeat 当前字符重复次数
        private int dfs(int n, List[] graph, String labels, int k, int cur, int sum, int repeat, boolean[] visited) {
            if (cur == n - 1) {
                return sum;
            }
            List<int[]> nextes = graph[cur];
            int min = Integer.MAX_VALUE;
            for (int[] next : nextes) {
                if (visited[next[0]]) {
                    continue;
                }
                if (labels.charAt(next[0]) == labels.charAt(cur)) {
                    if (repeat < k) {
                        visited[next[0]] = true;
                        min = Math.min(min, dfs(n, graph, labels, k, next[0], sum + next[1], repeat + 1, visited));
                        visited[next[0]] = false;
                    }
                } else {
                    visited[next[0]] = true;
                    min = Math.min(min, dfs(n, graph, labels, k, next[0], sum + next[1], 1, visited));
                    visited[next[0]] = false;
                }
            }
            return min;
        }
    }
```

```java
    class Solution {
        public int shortestPath(int n, int[][] edges, String labels, int k) {
            List[] graph = new List[n];
            for (int i = 0; i < n; i++) {
                graph[i] = new ArrayList<>();
            }
            for (int[] edge : edges) {
                graph[edge[0]].add(new int[]{edge[1], edge[2]});
            }
            // visited[i] 表示是否访问过i节点, 用例中有环
            boolean[] visited = new boolean[n];
            // 先根据cost排序, 如果cost相同, 那么再根据labelRepeat排序.
            // 相同节点因为不同路径被加入到pq, 取cost小的, 如果cost相同, 取字符重复次数少的
            PriorityQueue<Node> pq = new PriorityQueue<>(new Comparator<Node>() {
                @Override
                public int compare(Node o1, Node o2) {
                    if (o1.cost == o2.cost) {
                        return o1.labelRepeat - o2.labelRepeat;
                    }
                    return o1.cost - o2.cost;
                }
            });
            pq.add(new Node(0, 0, 1));
            while (!pq.isEmpty()) {
                Node poll = pq.poll();
                if (poll.id == n-1) {
                    return poll.cost;
                }
                if (visited[poll.id]) {
                    continue;
                }
                // 不能在此处将visited[poll.id]置为true
                // 只有在能继续下去, 即下个节点添加到pq时才把当前节点置为true
                // 不过我还是感觉这个剪枝逻辑有问题
                List<int[]> nextes = graph[poll.id];
                for (int[] next : nextes) {
                    if (labels.charAt(poll.id) == labels.charAt(next[0])) {
                        if (poll.labelRepeat == k) {
                            continue;
                        }
                        // 重复次数+1
                        pq.add(new Node(next[0], poll.cost + next[1], poll.labelRepeat + 1));
                        visited[poll.id] = true;
                    } else {
                        // 重复次数为1
                        pq.add(new Node(next[0], poll.cost + next[1], 1));
                        visited[poll.id] = true;
                    }
                }
            }
            return -1;
        }


        class Node {
            // 节点id
            public int id;
            // 访问该节点, 需要的cost
            public int cost;
            // 访问该节点,和该节点label相同字符的重复次数
            public int labelRepeat;

            public Node(int id, int cost, int labelRepeat) {
                this.id = id;
                this.cost = cost;
                this.labelRepeat = labelRepeat;
            }
        }
    }
```

让ai帮我改了一下, 用long[][] dist进行剪枝. 确实一开始的方案是有问题的, cost和labelRepeat加起来才能作为剪枝依据
```java
    class Solution {
        public int shortestPath(int n, int[][] edges, String labels, int k) {
            List[] graph = new List[n];
            for (int i = 0; i < n; i++) {
                graph[i] = new ArrayList<>();
            }
            for (int[] edge : edges) {
                graph[edge[0]].add(new int[]{edge[1], edge[2]});
            }
            // dist[u][c]: 到达u且末尾有c个连续相同标签时的最小代价, c取值[1,k]
            long[][] dist = new long[n][k + 1];
            for (long[] row : dist) {
                Arrays.fill(row, Long.MAX_VALUE);
            }
            // 先根据cost排序, 如果cost相同, 那么再根据labelRepeat排序.
            // 相同节点因为不同路径被加入到pq, 取cost小的, 如果cost相同, 取字符重复次数少的
            PriorityQueue<Node> pq = new PriorityQueue<>(new Comparator<Node>() {
                @Override
                public int compare(Node o1, Node o2) {
                    if (o1.cost == o2.cost) {
                        return o1.labelRepeat - o2.labelRepeat;
                    }
                    return Long.compare(o1.cost, o2.cost);
                }
            });
            dist[0][1] = 0;
            pq.add(new Node(0, 0, 1));
            while (!pq.isEmpty()) {
                Node poll = pq.poll();
                // 该状态已被更小的cost取代, 丢弃
                if (poll.cost > dist[poll.id][poll.labelRepeat]) {
                    continue;
                }
                if (poll.id == n - 1) {
                    return (int) poll.cost;
                }
                List<int[]> nextes = graph[poll.id];
                for (int[] next : nextes) {
                    // 标签相同则重复次数+1, 否则重置为1
                    int labelRepeat = labels.charAt(poll.id) == labels.charAt(next[0]) ? poll.labelRepeat + 1 : 1;
                    if (labelRepeat > k) {
                        continue;
                    }
                    long cost = poll.cost + next[1];
                    if (cost < dist[next[0]][labelRepeat]) {
                        dist[next[0]][labelRepeat] = cost;
                        pq.add(new Node(next[0], cost, labelRepeat));
                    }
                }
            }
            return -1;
        }


        class Node {
            // 节点id
            public int id;
            // 访问该节点, 需要的cost
            public long cost;
            // 访问该节点,和该节点label相同字符的重复次数
            public int labelRepeat;

            public Node(int id, long cost, int labelRepeat) {
                this.id = id;
                this.cost = cost;
                this.labelRepeat = labelRepeat;
            }
        }
    }
```
