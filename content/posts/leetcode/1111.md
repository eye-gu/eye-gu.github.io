    class Solution {
        public int[] maxDepthAfterSplit(String seq) {
            int aDepth = 0;
            int bDepth = 0;
            int[] result = new int[seq.length()];
            for (int i = 0; i < seq.length(); i++) {
                char c = seq.charAt(i);
                if (c == '(') {
                    // 左括号, 哪边深，就放另一边
                    if (aDepth < bDepth) {
                        aDepth++;
                        result[i] = 0;
                    } else {
                        bDepth++;
                        result[i] = 1;
                    }
                } else {
                    // 右括号, 哪边深，就放哪边
                    if (aDepth > bDepth) {
                        aDepth--;
                        result[i] = 0;
                    } else {
                        bDepth--;
                        result[i] = 1;
                    }
                }
            }
            return result;
        }
    }
