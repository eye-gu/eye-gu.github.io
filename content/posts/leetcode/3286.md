class Solution {
        public boolean findSafeWalk(List<List<Integer>> grid, int health) {
            int m = grid.size();
            int n = grid.get(0).size();
            if (grid.get(0).get(0) == 1) {
                health -= 1;
            }
            if (health >= m + n - 1) {
                return true;
            }
            int[][] dirs = new int[][] {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
            // 记录每个位置上最大的血量
            int[][] visited = new int[m][n];
            visited[0][0] = health;
            Queue<int[]> queue = new ArrayDeque<>();
            queue.offer(new int[] {0, 0, health});
            // bfs
            while (!queue.isEmpty()) {
                int[] poll = queue.poll();
                int x = poll[0];
                int y = poll[1];
                if (x == m - 1 && y == n - 1) {
                    return true;
                }
                int h = poll[2];
                for (int[] dir : dirs) {
                    int nx = x + dir[0];
                    int ny = y + dir[1];
                    if (nx < 0 || nx >= m || ny < 0 || ny >= n) {
                        continue;
                    }
                    if (grid.get(nx).get(ny) == 0) {
                        if (visited[nx][ny] < h) {
                            queue.offer(new int[] {nx, ny, h});
                            visited[nx][ny] = h;
                        }
                    } else {
                        if (h > 1) {
                            if (visited[nx][ny] < h - 1) {
                                queue.offer(new int[] {nx, ny, h - 1});
                                visited[nx][ny] = h - 1;
                            }
                        }
                    }
                }
            }
            return false;
        }
    }

// Dijkstra
    class Solution {
        public boolean findSafeWalk(List<List<Integer>> grid, int health) {
            int m = grid.size();
            int n = grid.get(0).size();
            if (health >= m + n) {
                return true;
            }
            // 所有可达的点,分0和1,永远是值为0的点先取. 只有0没有的时候,才走一次1
            ArrayDeque<int[]> zeroPoints = new ArrayDeque<>();
            ArrayDeque<int[]> onePoints = new ArrayDeque<>();
            int cur = 0;
            int[][] dirs = {{0, 1}, {0, -1}, {1, 0}, {-1, 0}};
            boolean[][] visited = new boolean[m][n];
            if (grid.get(0).get(0) == 1) {
                onePoints.offer(new int[]{0, 0});
            } else {
                zeroPoints.offer(new int[]{0, 0});
            }
            while (cur < health) {
                if (!zeroPoints.isEmpty()) {
                    int size = zeroPoints.size();
                    for (int i =0; i < size; i++) {
                        int[] point = zeroPoints.poll();
                        if (point[0] == m - 1 && point[1] == n - 1) {
                            return true;
                        }
                        if (visited[point[0]][point[1]]) {
                            continue;
                        }
                        visited[point[0]][point[1]] = true;
                        for (int[] dir : dirs) {
                            int nx = point[0] + dir[0];
                            int ny = point[1] + dir[1];
                            if (nx >= 0 && nx < m && ny >= 0 && ny < n && !visited[nx][ny]) {
                                if (grid.get(nx).get(ny) == 1) {
                                    onePoints.offer(new int[]{nx, ny});
                                } else {
                                    zeroPoints.offer(new int[]{nx, ny});
                                }
                            }
                        }
                    }
                } else {
                    cur++;
                    int size = onePoints.size();
                    for (int i =0; i < size; i++) {
                        int[] point = onePoints.poll();
                        if (point[0] == m - 1 && point[1] == n - 1) {
                            if (cur < health) {
                                return true;
                            } else {
                                return false;
                            }
                        }
                        if (visited[point[0]][point[1]]) {
                            continue;
                        }
                        visited[point[0]][point[1]] = true;
                        for (int[] dir : dirs) {
                            int nx = point[0] + dir[0];
                            int ny = point[1] + dir[1];
                            if (nx >= 0 && nx < m && ny >= 0 && ny < n && !visited[nx][ny]) {
                                if (grid.get(nx).get(ny) == 1) {
                                    onePoints.offer(new int[]{nx, ny});
                                } else {
                                    zeroPoints.offer(new int[]{nx, ny});
                                }
                            }
                        }
                    }
                }
            }
            return false;
        }
    }
