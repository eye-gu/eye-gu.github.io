// 关键的思路还是官解给的,所有值排序,相邻间隔大于limit,就切分开,每组内可以随意更改位置
    class Solution {
        public int[] lexicographicallySmallestArray(int[] nums, int limit) {
            int n = nums.length;
            // 索引,根据值排序
            Integer[] indexes = new Integer[n];
            for (int i =0; i<n; i++) {
                indexes[i] = i;
            }
            Arrays.sort(indexes, Comparator.comparingInt(o -> nums[o]));

            int[] ans = new int[n];
            int i =0;
            while (i < n) {
                List<Integer> temps = new ArrayList<>();
                List<Integer> tempIndexed = new ArrayList<>();

                temps.add(nums[indexes[i]]);
                tempIndexed.add(indexes[i]);

                // 收集所有本组内的值和索引
                while (i < n -1 && nums[indexes[i + 1]] - nums[indexes[i]] <= limit) {
                    temps.add(nums[indexes[i+1]]);
                    tempIndexed.add(indexes[i+1]);
                    i++;
                }
                // 只需对索引排序,值已经排序
                Collections.sort(tempIndexed);
                // 根据索引从小到大,将值填充到ans中
                for (int j = 0; j < temps.size(); j++) {
                    ans[tempIndexed.get(j)] = temps.get(j);
                }
                i++;
            }
            return ans;
        }
    }
