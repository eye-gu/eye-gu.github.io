这个思路讲道理没那么难，可是一开始还是没想出来，在ai的提示下才写出来

```java
    class Solution {
        public List<Integer> busiestServers(int k, int[] arrival, int[] load) {
            // idleTime，serverId
            PriorityQueue<int[]> pq = new PriorityQueue<>(Comparator.comparingInt(a -> a[0]));
            // 目前空闲的服务器id
            TreeSet<Integer> idleServers = new TreeSet<>();
            for (int i = 0; i < k; i++) {
                idleServers.add(i);
            }
            int[] times = new int[k];
            for (int i =0; i<arrival.length; i++) {
                int arrivalTime = arrival[i];

                // 处理所有空闲的服务器
                while (!pq.isEmpty() && pq.peek()[0] <= arrivalTime) {
                    int[] poll = pq.poll();
                    idleServers.add(poll[1]);
                }

                if (idleServers.isEmpty()) {
                    continue;
                }

                Integer serverId = idleServers.ceiling(i % k);
                // 环
                if (serverId == null) {
                    serverId = idleServers.first();
                }
                times[serverId]++;
                idleServers.remove(serverId);
                pq.add(new int[]{arrivalTime + load[i], serverId});
            }
            int maxTime = Arrays.stream(times).max().getAsInt();
            return IntStream.range(0, k).filter(i -> times[i] == maxTime).boxed().toList();
        }
    }
```
