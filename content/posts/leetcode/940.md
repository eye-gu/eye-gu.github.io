class Solution {
        public int distinctSubseqII(String s) {
            final int MOD = 1000000007;
            // last[i] 表示字符 i 上一次出现的下标
            int[] last = new int[26];
            Arrays.fill(last, -1);

            int n = s.length();
            // f[i] 表示以 s[i] 结尾的子序列的数量
            // 关键: 如果i<j, s[i] = s[j], 那么 f[i] 的所有的子序列,都包含在 f[j] 中
            // 就感觉是把子序列分成了26种,分别是以a,b,c...z结尾的子序列. f[i]就等于当前26种子字符串的和,加上自己单独作为子字符串的1, 最后把s[i]字符结尾,指向自己
            int[] f = new int[n];
            Arrays.fill(f, 1);
            for (int i = 0; i < n; ++i) {
                for (int j = 0; j < 26; ++j) {
                    if (last[j] != -1) {
                        f[i] = (f[i] + f[last[j]]) % MOD;
                    }
                }
                last[s.charAt(i) - 'a'] = i;
            }

            int ans = 0;
            for (int i = 0; i < 26; ++i) {
                if (last[i] != -1) {
                    ans = (ans + f[last[i]]) % MOD;
                }
            }
            return ans;
        }
    }
