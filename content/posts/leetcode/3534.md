https://leetcode.cn/problems/path-existence-queries-in-a-graph-ii/description/?envType=daily-question&envId=2026-07-10



因为maxRight一步一步走,会超时

```java
class Solution {
        public int[] pathExistenceQueries(int n, int[] nums, int maxDiff, int[][] queries) {
            // nums从小到大排序的序号
            Integer[] index = new Integer[n];
            for (int i =0; i<n; i++) {
                index[i] = i;
            }
            Arrays.sort(index, Comparator.comparingInt(a -> nums[a]));

            // maxRight[i]表示节点i走一步,最远可以到达的节点
            int[] maxRight = new int[n];
            // tags[i]表示nums[i]所在的组的编号, tags[i] = tags[j] 表示nums[i]和nums[j]连通
            int[] tags = new int[n];
            int left = 0;
            for (int i =1; i<n; i++) {
                if (nums[index[i]] - nums[index[i-1]] <= maxDiff) {
                    tags[index[i]] = tags[index[i-1]];
                } else {
                    tags[index[i]] = tags[index[i-1]] + 1;
                }
                while (nums[index[i]] - nums[index[left]] > maxDiff) {
                    maxRight[index[left]] = index[i-1];
                    left++;
                }
            }
            for (int i=left; i<n; i++) {
                maxRight[index[i]] = index[n-1];
            }
            int[] res = new int[queries.length];
            for (int i=0; i<queries.length; i++) {
                int[] query = queries[i];
                if (tags[query[0]] != tags[query[1]]) {
                    res[i] = -1;
                } else {
                    if (nums[query[0]] == nums[query[1]]) {
                        if (query[0] == query[1]) {
                            res[i] = 0;
                        } else {
                            res[i] = 1;
                        }
                        continue;
                    }
                    int cur = query[0];
                    int end = query[1];
                    if (nums[query[0]] > nums[query[1]]) {
                        cur = query[1];
                        end = query[0];
                    }
                    int count = 0;
                    while (nums[cur] < nums[end]) {
                        count++;
                        cur = maxRight[cur];
                    }
                    res[i] = count;
                }
            }
            return res;
        }
    }
```
