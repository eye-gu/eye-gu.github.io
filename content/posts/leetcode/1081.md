```java
class Solution {
        public String smallestSubsequence(String s) {
            int[] count = new int[26];
            for (char c : s.toCharArray()) {
                count[c - 'a']++;
            }
            StringBuilder stack = new StringBuilder();
            // used[i] 表示字符 i 是否在栈中
            boolean[] used = new boolean[26];
            for (char c : s.toCharArray()) {
                count[c - 'a']--;
                if (used[c - 'a']) {
                    continue;
                }
                // peek: last char; push: append; pop: deleteLastChar
                if (!stack.isEmpty()) {
                    char top = stack.charAt(stack.length() - 1);
                    if (top >= c) {
                        // count[top - 'a'] > 0 表示 top 在后续字符串中还存在,如果不存在了,那就不能被移除
                        while (!stack.isEmpty() && (top = stack.charAt(stack.length() - 1)) > c && count[top - 'a'] > 0) {
                            used[top - 'a'] = false;
                            stack.deleteCharAt(stack.length() - 1);
                        }
                    }
                }
                stack.append(c);
                used[c - 'a'] = true;
            }
            return stack.toString();
        }
    }
```
