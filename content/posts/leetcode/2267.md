```java
    class Solution {
        public boolean hasValidPath(char[][] grid) {
            int n = grid.length;
            int m = grid[0].length;
            if (grid[0][0] != '(' || grid[n - 1][m - 1] != ')') {
                return false;
            }
            // 通过memo对dfs剪枝
            Set<Integer> memo = new HashSet<>();
            return dfs(grid, 0, 0, 1, memo);
        }

        private boolean dfs(char[][] grid, int i, int j, int left, Set<Integer> memo) {
            if (i == grid.length - 1 && j == grid[0].length - 1) {
                return left == 0;
            }
            int key = i * 10000 + j * 100 + left;
            if (memo.contains(key)) {
                return false;
            }
            memo.add(key);
            // 往右
            if (i < grid.length - 1) {
                if (grid[i+1][j] == '(') {
                    if (dfs(grid, i + 1, j, left + 1, memo)) {
                        return true;
                    }
                } else {
                    if (left > 0) {
                        if (dfs(grid, i + 1, j, left - 1, memo)) {
                            return true;
                        }
                    }
                }
            }
            // 往下
            if (j < grid[0].length - 1) {
                if (grid[i][j+1] == '(') {
                    if (dfs(grid, i, j + 1, left + 1, memo)) {
                        return true;
                    }
                } else {
                    if (left > 0) {
                        if (dfs(grid, i, j + 1, left - 1, memo)) {
                            return true;
                        }
                    }
                }
            }
            return false;
        }
    }
```

```java
    class Solution {
        public boolean hasValidPath(char[][] grid) {
            int n = grid.length;
            int m = grid[0].length;
            int pathLen = n + m - 1;

            if (pathLen % 2 == 1) {
                return false;
            }
            if (grid[0][0] != '(' || grid[n - 1][m - 1] != ')') {
                return false;
            }
            // dp[i][j][k] 代表路径到 grid[i][j] 时, 左括号减右括号的数量差为 k 是否存在
            boolean[][][] dp = new boolean[n][m][pathLen + 1];
            // 起点只存在差值为1
            dp[0][0][1] = true;

            for (int i = 0; i < n; ++i) {
                for (int j = 0; j < m; ++j) {
                    int change = grid[i][j] == '(' ? 1 : -1;

                    if (i > 0) {
                        // 从grid[i-1][j]移动到grid[i][j], 左括号减右括号的数量差为balance存在, 则grid[i][j]的左括号减右括号的数量差为balance+change存在
                        for (int balance = 0; balance <= pathLen; ++balance) {
                            if (!dp[i - 1][j][balance]) {
                                continue;
                            }

                            int next = balance + change;

                            if (next >= 0) {
                                dp[i][j][next] = true;
                            }
                        }
                    }

                    if (j > 0) {
                        // 从grid[i][j-1]移动到grid[i][j], 左括号减右括号的数量差为balance存在, 则grid[i][j]的左括号减右括号的数量差为balance+change存在
                        for (int balance = 0; balance <= pathLen; ++balance) {
                            if (!dp[i][j - 1][balance]) {
                                continue;
                            }

                            int next = balance + change;

                            if (next >= 0) {
                                dp[i][j][next] = true;
                            }
                        }
                    }
                }
            }

            return dp[n - 1][m - 1][0];
        }
    }
```
